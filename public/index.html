    const int PIN_X  = A0;
const int PIN_Y  = A1;
const int PIN_SW = 2;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_SW, INPUT_PULLUP);
}

void loop() {
  int x = analogRead(PIN_X);
  int y = analogRead(PIN_Y);
  int b = (digitalRead(PIN_SW) == LOW) ? 1 : 0;
  Serial.print(x); Serial.print(',');
  Serial.print(y); Serial.print(',');
  Serial.println(b);
  delay(20);
}
